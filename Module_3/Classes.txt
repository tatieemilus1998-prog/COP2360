using System.Diagnostics;

public class Bunny
{
    public string Name;
    public bool EnjoysCarrots;
    public bool LikesHumans;

    public Bunny(
      string name,
      bool enjoysCarrots = false,
      bool likesHumans = false)
    {
        Name = name;
        EnjoysCarrots = enjoysCarrots;
        LikesHumans = likesHumans;
    }

    public void Output()
    {
        string message = Name + " is a bunny that does ";

        if (!EnjoysCarrots)
        {
            message += "not ";
        }

        message += "enjoy carrots and also does ";

        if (!LikesHumans)
        {
            message += "not ";
        }

        message += "like humans";

        Console.WriteLine(message);
    }
}

public class aProgram
{ 
    static void Main()
    {
        Console.WriteLine("Name your new bunny");
        string? desiredName = Console.ReadLine();

        if (desiredName == null)
        {
            desiredName = "Just a bunny";
        }

        Console.WriteLine("Does your bunny enjoy carrots? yes/no");
        string? response = Console.ReadLine();

        Debug.Assert(response != null, "response must not be null");

        bool enjoysCarrots;

        if (response == "yes" || response == "Yes")
        {
            enjoysCarrots = true;
        }
        else
        {
            enjoysCarrots = false;
        }

        Console.WriteLine("Does your bunny like humans? yes/no");
        string? response2 = Console.ReadLine();

        Debug.Assert(response2 != null, "response must not be null");

        bool likesHumans;

        if (response2 == "yes" || response2 == "Yes")
        {
            likesHumans = true;
        }
        else
        {
            likesHumans = false;
        }

        Bunny newBunny = new Bunny(desiredName, enjoysCarrots, likesHumans);
        newBunny.Output();
    }
}
